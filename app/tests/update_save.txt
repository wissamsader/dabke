# An update starts (the computer's upgrade command, from the phone's firmware switch): whatever DABKE has not saved yet
# (the settings, the sequencer's part) is saved at that moment, whatever plays, so it is there when DABKE is put back.
# (The save otherwise waits for a quiet moment, a few seconds after the last change.) "update" is the start of the
# session, "reboot" the install's restart; "hash_keep" then "expect hash_same 1": everything kept came back the same.
# From the repository's root:
#   app/build/host/dabke_host app/tests/update_save.txt OUTDIR
wait 900
expect store_writes 0
# nothing unsaved: nothing written
update
expect store_writes 0
expect store_tries 0
wait 300
# a change made while the rhythm runs: saved at the update, and there after the restart
tap PLAY
wait 200
expect running 1
turn ALGO 2
wait 5000
expect tempo 144
expect running 1
expect dirty 1
expect store_writes 0
update
expect running 0
expect dirty 0
expect store_writes 1
hash_keep
reboot
wait 900
expect tempo 144
expect dirty 0
expect store_writes 1
expect hash_same 1
# a change less than three seconds old, the rhythm stopped: saved at the update
turn ALGO 1
wait 500
expect tempo 146
expect dirty 1
expect store_writes 1

expect dirty 0
expect store_writes 2
reboot
wait 900
expect tempo 146
